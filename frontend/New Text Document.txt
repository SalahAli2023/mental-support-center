Options -MultiViews

<IfModule mod_rewrite.c>
RewriteEngine On
RewriteBase /

# =========================
# 1️⃣  لو الرابط يبدأ بـ /api
# يروح إلى Laravel (index.php)
# =========================
RewriteCond %{REQUEST_URI} ^/api
RewriteRule ^ index.php [L]

# =========================
# 2️⃣  لو الملف أو المجلد موجود فعلياً
# لا تعد كتابة الرابط
# =========================
RewriteCond %{REQUEST_FILENAME} -f [OR]
RewriteCond %{REQUEST_FILENAME} -d
RewriteRule ^ - [L]

# =========================
# 3️⃣  باقي الروابط تروح إلى Vue
# =========================
RewriteRule ^ index.html [L]
</IfModule>
